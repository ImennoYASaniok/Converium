// ER-диаграмма dbdiagram.io

// Перечисления (ENUM)
Enum visibility_enum {
  PUBLIC
  FRIENDS_ONLY
  CERTAIN_PEOPLE
}

Enum moderation_status_enum {
  PENDING
  APPROVED
  REJECTED
}

Enum ability_enum {
  READ
  EDIT
  ADMIN
}

Enum step_type_enum {
  THEORY
  TEST
  PRACTICE
}

Enum question_type_enum {
  TEXT_INPUT
  SINGLE_CHOICE
  MULTIPLE_CHOICE
}

Enum contact_type_enum {
  PHONE
  TELEGRAM
  EMAIL
  OTHER
}

// Пользователи
Table users {
  id bigint [pk, increment]
  login varchar(40) [unique, not null]
  password varchar(255) [not null]
  email varchar(50) [unique, not null]
  profile_picture varchar(255) [default: ""]
  name varchar(30) [default: ""]
  surname varchar(30) [default: ""]
  description text [default: ""]
  created_at timestamp [not null, default: `now()`]
}

// Контакты пользователя
Table user_contacts {
  user_id bigint [not null, ref: > users.id]
  contact_type contact_type_enum [not null]
  contact varchar(255) [not null]
  indexes {
    (user_id, contact) [unique]
  }
}

// Друзья (many-to-many) с проверкой упорядочивания
Table user_friends {
  user_id bigint [not null, ref: > users.id]
  friend_id bigint [not null, ref: > users.id]
  indexes {
    (user_id, friend_id) [pk]
  }
}

// Баны
Table user_bans {
  user_id bigint [not null, ref: > users.id]
  banned_user_id bigint [not null, ref: > users.id]
  indexes {
    (user_id, banned_user_id) [pk]
  }
}

// Запросы в друзья
Table friend_requests {
  user_id bigint [not null, ref: > users.id] // получатель
  requester_id bigint [not null, ref: > users.id] // отправитель
  indexes {
    (user_id, requester_id) [pk]
  }
}

// Курсы
Table courses {
  id bigint [pk, increment]
  title varchar(200) [not null]
  description text
  created_at timestamp [not null, default: `now()`]
  updated_at timestamp [not null, default: `now()`]
  owner_id bigint [not null, ref: > users.id]
  visibility visibility_enum [not null, default: 'FRIENDS_ONLY']
  moderation_status moderation_status_enum
  comments_enabled boolean [not null, default: true]
}

// Записи на курс
Table course_enrollments {
  id bigint [pk, increment]
  course_id bigint [not null, ref: > courses.id]
  user_id bigint [not null, ref: > users.id]
  enrolled_at timestamp [not null, default: `now()`]
  indexes {
    (course_id, user_id) [unique]
  }
}

// Должности в курсе (права доступа)
Table course_memberships {
  id bigint [pk, increment]
  course_id bigint [not null, ref: > courses.id]
  user_id bigint [not null, ref: > users.id]
  ability ability_enum [not null, default: 'READ']
  granted_by bigint [ref: > users.id]
  indexes {
    (course_id, user_id) [unique]
  }
}

// Шаги курса
Table steps {
  id bigint [pk, increment]
  name varchar(255) [not null]
  content text
  course_id bigint [not null, ref: > courses.id]
  type step_type_enum [not null, default: 'THEORY']
}

// Рёбра графа шагов
Table step_edges {
  id bigint [pk, increment]
  from_step_id bigint [not null, ref: > steps.id]
  to_step_id bigint [not null, ref: > steps.id]
  required_score int
  course_id bigint [not null, ref: > courses.id]
  indexes {
    (from_step_id, to_step_id) [unique]
  }
}

// Приватные чаты (для редакторов курса)
Table private_chat_rooms {
  id bigint [pk, increment]
  course_id bigint [ref: > courses.id]
  step_id bigint [ref: > steps.id]
  name varchar(200) [not null]
  created_at timestamp [not null, default: `now()`]
}

// Участники приватных чатов
Table private_chat_participants {
  id bigint [pk, increment]
  chat_room_id bigint [not null, ref: > private_chat_rooms.id]
  user_id bigint [not null, ref: > users.id]
  joined_at timestamp [not null, default: `now()`]
  last_read_message_id bigint [ref: > private_chat_messages.id]
  indexes {
    (chat_room_id, user_id) [unique]
  }
}

// Сообщения приватных чатов
Table private_chat_messages {
  id bigint [pk, increment]
  chat_room_id bigint [not null, ref: > private_chat_rooms.id]
  user_id bigint [not null, ref: > users.id]
  content text [not null]
  sent_at timestamp [not null, default: `now()`]
  updated_at timestamp
  reply_to_message_id bigint [ref: > private_chat_messages.id]
  indexes {
    (chat_room_id, sent_at)
  }
}

// Комментарии к курсу или шагу
Table comments {
  id bigint [pk, increment]
  author_id bigint [not null, ref: > users.id]
  course_id bigint [ref: > courses.id]
  step_id bigint [ref: > steps.id]
  content text [not null]
  created_at timestamp [not null, default: `now()`]
  parent_comment_id bigint [ref: > comments.id]
  indexes {
    (course_id, created_at)
    (step_id, created_at)
  }
}

// Доски (Miro‑подобные)
Table boards {
  id bigint [pk, increment]
  step_id bigint [unique, not null, ref: > steps.id]
  data json [not null]
  created_at timestamp [not null, default: `now()`]
  updated_at timestamp [not null, default: `now()`]
  version int [not null, default: 0]
}

// Тесты
Table tests {
  id bigint [pk, increment]
  step_id bigint [unique, not null, ref: > steps.id]
  time_limit_minutes int
  attempts_allowed int [default: 1]
  created_at timestamp [not null, default: `now()`]
  updated_at timestamp [not null, default: `now()`]
}

// Вопросы теста
Table questions {
  id bigint [pk, increment]
  test_id bigint [not null, ref: > tests.id]
  text text [not null]
  type question_type_enum [not null]
  points int [not null, default: 1]
  order_index int [not null]
  created_at timestamp [not null, default: `now()`]
  updated_at timestamp [not null, default: `now()`]
  correct_text text
  indexes {
    (test_id, order_index)
  }
}

// Варианты ответов для вопросов с выбором
Table choice_questions {
  id bigint [pk, increment]
  question_id bigint [not null, ref: > questions.id]
  order_number int [not null]
  text text
  is_correct boolean
  indexes {
    (question_id)
  }
}

// Попытки прохождения теста
Table test_attempts {
  id bigint [pk, increment]
  test_id bigint [not null, ref: > tests.id]
  user_id bigint [not null, ref: > users.id]
  attempt_number int [not null]
  started_at timestamp [not null, default: `now()`]
  completed_at timestamp
  score int
  indexes {
    (test_id, user_id, attempt_number) [unique]
    (user_id, started_at)
  }
}

// Ответы пользователя (общая часть)
Table user_answers {
  id bigint [pk, increment]
  test_attempt_id bigint [not null, ref: > test_attempts.id]
  question_id bigint [not null, ref: > questions.id]
  type question_type_enum [not null]
  text text
  is_correct boolean
  indexes {
    (test_attempt_id, question_id) [unique]
  }
}

// Связь ответов с выбранными вариантами (для SINGLE_CHOICE / MULTIPLE_CHOICE)
Table choice_user_answers {
  user_answer_id bigint [not null, ref: > user_answers.id]
  choice_question_id bigint [not null, ref: > choice_questions.id]
  indexes {
    (user_answer_id, choice_question_id) [pk]
  }
}
